               coalesce(sum(c.qtd_visualizacoes), 0)::BIGINT AS n_visualizacoes
        FROM steam_do_bedo.plataforma p

        GROUP BY p.nro, p.nome, p.qtd_users
    ),
    ranking AS (
        -- posição global, ANTES do filtro: o filtro não muda a posição
